  // СТОЯНИЕ: чего вы стоите в их своде к последней главе
  //
  // Не «очки добра». Это счёт ровно по тем правилам, которыми держава живёт
  // всю хронику: регулярность вместо щедрости (§7.2), заполненное умолчание
  // вместо удобного (§11.2), покой вместо силы (§8.1), долгая дорога к ихору
  // вместо вскрытой (§15). Финал спрашивает не «что вы выбрали в конце»,
  // а «кто вы для них после пяти глав».
  // ══════════════════════════════════════════════════════════
  function stand(f) {
    f = f || {};
    let n = 0;
    n += f.уговор === 'ряд' ? 2 : f.уговор === 'разом' ? 0 : -1;
    n += f.летопись === 'правда' ? 2 : f.летопись === 'полуправда' ? 0 : f.летопись === 'подлог' ? -1 : 0;
    n += (+f.покой || 0);
    n -= f.страх ? 1 : 0;
    n += f.ихор === 'долгая' ? 2 : f.ихор === 'вскрыл' ? (f.откуп ? 0 : -1) : 0;
    return n;
  }
